function runSimulation() { if (baselineLiveIncome === 0) return; const adBudget = parseInt(document.getElementById('sim-ad-slider').value); const envScore = parseInt(document.getElementById('sim-env-slider').value); const effScore = parseInt(document.getElementById('sim-eff-slider').value); const targetStars = parseInt(document.getElementById('sim-star-slider').value); const hasTGP = document.getElementById('sim-tgp-stock').checked; const hasAdEdu = document.getElementById('sim-ad-edu').checked; document.getElementById('sim-ad-val').innerText = `$${adBudget.toLocaleString()}`; document.getElementById('sim-env-val').innerText = `${envScore}%`; document.getElementById('sim-eff-val').innerText = `${effScore}%`; const starDiff = targetStars - liveStarRating; if (starDiff === 0) document.getElementById('sim-star-val').innerText = `Current Stars (${targetStars})`; else if (starDiff > 0) document.getElementById('sim-star-val').innerText = `+${starDiff} Stars`; else document.getElementById('sim-star-val').innerText = `${starDiff} Stars`; let projectedDaily; if (companyHistory.length >= 5) { const recent = companyHistory.slice(-7); const avg = recent.reduce((a,b)=>a+b,0) / recent.length; let trend = 0; for (let i = 1; i < recent.length; i++) { trend += recent[i] - recent[i-1]; } trend /= (recent.length - 1); projectedDaily = avg + trend; } else { projectedDaily = baselineLiveIncome; } projectedDaily *= (effScore / 100); const envPenalty = ((100 - envScore) / 100) * 0.33; projectedDaily *= (1 - envPenalty); let adBoost = Math.min( 0.40, (adBudget / 1000000) * 0.40 ); if (hasTGP) adBoost *= 1.15; if (hasAdEdu) adBoost *= 1.10; projectedDaily *= (1 + adBoost); const starBoost = 1 + ((targetStars - liveStarRating) * 0.08); projectedDaily *= starBoost; const projected30Day = projectedDaily * 30; let confidence = 70; if (companyHistory.length >= 7) { const avg = companyHistory.reduce((a,b)=>a+b,0) / companyHistory.length; const variance = companyHistory.reduce( (s,v)=>s+Math.pow(v-avg,2), 0 ) / companyHistory.length; const volatility = Math.sqrt(variance); confidence = Math.max( 45, Math.min( 95, 95 - ((volatility/avg)*100) ) ); } const bestCase = projected30Day * 1.12; const worstCase = projected30Day * 0.88; document.getElementById('sim-output-revenue') .innerText = '$' + Math.round(projected30Day) .toLocaleString(); let recommendation = "Revenue stable."; if (envScore < 80) recommendation = "Environment score is hurting profits."; else if (effScore < 90) recommendation = "Employee effectiveness is the main bottleneck."; else if (adBudget < 250000) recommendation = "Advertising budget has room for growth."; document.getElementById( 'sim-output-change' ).innerHTML = `
Confidence: ${confidence.toFixed(0)}%
Best Case: $${Math.round(bestCase).toLocaleString()}
Worst Case: $${Math.round(worstCase).toLocaleString()}
${recommendation}
`; }
